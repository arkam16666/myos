# MyOS for Raspberry Pi Zero 2 W, booted over USB by rpiboot
arm_64bit=1
kernel=kernel8.img
# small "cut-down" firmware (~800 KB instead of 3 MB): quicker over USB,
# and MyOS doesn't use the GPU's extra features
start_file=start_cd.elf
fixup_file=fixup_cd.dat
# serial console on GPIO 14 (TX) / 15 (RX), 115200 baud
enable_uart=1
dtoverlay=disable-bt
init_uart_clock=48000000
